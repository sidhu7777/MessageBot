SUCCESS_TABLES=0
ATTEMPTED_TABLES=0
FAILED_TABLE=""

dump_table() {
  local table="$1"
  local where_clause="$2"

  mysqldump \
    --defaults-extra-file="$MYSQL_DEFAULTS_FILE" \
    --single-transaction \
    --skip-lock-tables \
    --no-tablespaces \
    --no-create-info \
    --skip-triggers \
    "$DB_NAME" \
    "$table" \
    --where="$where_clause" >> "$FILE"
}

run_dump() {
  local table="$1"
  local where_clause="$2"

  if [ -n "$FAILED_TABLE" ]; then
    return
  fi

  ATTEMPTED_TABLES=$((ATTEMPTED_TABLES + 1))

    SUCCESS_TABLES=$((SUCCESS_TABLES + 1))

    FAILED_TABLE="$table"

#!/bin/bash

set -u

export TZ=Asia/Kolkata

DATE=$(date +%F_%H-%M-%S)
BACKUP_DIR="/backup/mysql"
DB_NAME="doctor_db"
EMAILS="bvineeth76@gmail.com harshittripathi991@gmail.com"
MYSQL_DEFAULTS_FILE="/root/.my.cnf.bak"
FILE="$BACKUP_DIR/${DB_NAME}_full_${DATE}.sql"
ARCHIVE="$FILE.gz"

mkdir -p "$BACKUP_DIR"

mysqldump \
  --defaults-extra-file="$MYSQL_DEFAULTS_FILE" \
  --single-transaction \
  --skip-lock-tables \
  --no-tablespaces \
  "$DB_NAME" > "$FILE"

if [ $? -eq 0 ] && [ -s "$FILE" ]; then
  gzip "$FILE"

  if [ -s "$ARCHIVE" ]; then
    rclone copy "$ARCHIVE" gdrive:mysql-backups/

    if [ $? -eq 0 ]; then
      MESSAGE="SUCCESS: Full MySQL backup created and uploaded: $ARCHIVE"
    else
      MESSAGE="WARNING: Full MySQL backup created locally but Google Drive upload failed: $ARCHIVE"
    fi
  else
    MESSAGE="FAILED: Full backup compression failed"
  fi
else
  rm -f "$FILE"
  MESSAGE="FAILED: Full MySQL dump failed"
fi

echo "$MESSAGE"
echo "$MESSAGE" | mail -s "Full MySQL Backup Status" $EMAILS

find /backup/mysql -type f -name '*_full_*.sql.gz' -mtime +35 -delete
